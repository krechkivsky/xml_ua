            for feature in added_features:
                if not isinstance(feature, QgsFeature):
                    continue
                geometry = feature.geometry()
                if not geometry or geometry.isNull():
                    continue

                # 1. Якщо об'єкт у QGIS ВЖЕ має ID (наприклад, імпортований або повторно збережений) — пропускаємо
                existing_obj_id = _try_attribute(feature, "object_id")
                if existing_obj_id is not None and str(existing_obj_id).strip() not in ("", "0", "None"):
                    continue

                # --- Логіка для шару "Угіддя" ---
                if layer_name == "Угіддя":
                    if layer.geometryType() != QgsWkbTypes.PolygonGeometry:
                        continue
                    
                    externals, _, _, object_shape = processor.process_new_geometry(geometry)
                    if externals is None or not object_shape:
                        continue

                    clean_shape = str(object_shape).strip()

                    # 2. Перевірка ТІЛЬКИ в оперативній пам'яті (без XPath / без .find)
                    # Якщо ця геометрія вже була додана під час поточного циклу — пропускаємо
                    if any(si.layer_id == layer.id() and str(si.object_shape).strip() == clean_shape for si in xml_data.shapes):
                        continue

                    # Додавання нового угіддя
                    land_code = _safe_text(_try_attribute(feature, "LandCode"))
                    size_ha = float(_area_ha_text(geometry))
                    
                    object_id = processor.add_land_parcel_info(externals, land_code, size_ha, clean_shape)
                    _append_shape_info(object_id, clean_shape)
                    _stage_linkage_fields(feature.id(), object_id, clean_shape)
                    changed = True
